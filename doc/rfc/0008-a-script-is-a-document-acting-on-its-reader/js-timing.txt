## src/bin/boa.rs

    //! Construct a Boa context, evaluate `1+1`, and time both; then a runaway loop under RuntimeLimits.
    use std::time::Instant;
    use boa_engine::{Context, Source, JsValue};
    fn main() {
        let t0 = Instant::now();
        let mut context = Context::default();
        let t1 = Instant::now();
        let value = context.eval(Source::from_bytes("1+1")).expect("1+1");
        let t2 = Instant::now();
        assert_eq!(value, JsValue::from(2));
        let value2 = context.eval(Source::from_bytes("var s=0; for (var i=0;i<1000;i++) s+=i; s")).expect("loop");
        let t3 = Instant::now();
        assert_eq!(value2, JsValue::from(499500));
        println!("boa\tconstruct={:?}\teval(1+1)={:?}\teval(loop1000)={:?}", t1 - t0, t2 - t1, t3 - t2);
        // Second context, cold-ish again, to see construction variance.
        let t4 = Instant::now();
        let mut c2 = Context::default();
        let t5 = Instant::now();
        println!("boa\tconstruct#2={:?}", t5 - t4);
        // Budgets: loop-iteration and recursion limits, the two RuntimeLimits offers.
        let mut limits = c2.runtime_limits();
        limits.set_loop_iteration_limit(1_000_000);
        limits.set_recursion_limit(100);
        c2.set_runtime_limits(limits);
        let t6 = Instant::now();
        let r = c2.eval(Source::from_bytes("while(true){}"));
        let t7 = Instant::now();
        println!("boa\twhile(true) under loop limit: {:?} after {:?}", r.map(|_| "finished").map_err(|e| e.to_string()), t7 - t6);
        let t8 = Instant::now();
        let r = c2.eval(Source::from_bytes("function f(){return f();} f()"));
        let t9 = Instant::now();
        println!("boa\tunbounded recursion under recursion limit: {:?} after {:?}", r.map(|_| "finished").map_err(|e| e.to_string()), t9 - t8);
        // What RuntimeLimits does not bound: work inside a built-in.
        let t10 = Instant::now();
        let r = c2.eval(Source::from_bytes("var a = new Array(5000000).fill(0); a.length"));
        let t11 = Instant::now();
        println!("boa\tnew Array(5e6).fill(0): {:?} after {:?}", r.map(|v| v.display().to_string()).map_err(|e| e.to_string()), t11 - t10);
    }

## src/bin/quickjs.rs

    //! Construct a QuickJS runtime and context, evaluate `1+1`, and time both; then a runaway loop under an interrupt handler and a memory limit.
    use std::time::{Duration, Instant};
    use rquickjs::{Context, Runtime};
    fn main() {
        let t0 = Instant::now();
        let rt = Runtime::new().expect("runtime");
        let ctx = Context::full(&rt).expect("context");
        let t1 = Instant::now();
        let v: i32 = ctx.with(|ctx| ctx.eval("1+1")).expect("1+1");
        let t2 = Instant::now();
        assert_eq!(v, 2);
        let v2: i32 = ctx.with(|ctx| ctx.eval("var s=0; for (var i=0;i<1000;i++) s+=i; s")).expect("loop");
        let t3 = Instant::now();
        assert_eq!(v2, 499500);
        println!("quickjs\tconstruct={:?}\teval(1+1)={:?}\teval(loop1000)={:?}", t1 - t0, t2 - t1, t3 - t2);
        let t4 = Instant::now();
        let rt2 = Runtime::new().expect("runtime");
        let ctx2 = Context::full(&rt2).expect("context");
        let t5 = Instant::now();
        println!("quickjs\tconstruct#2={:?}", t5 - t4);
        // Budgets: a wall-clock interrupt and a memory ceiling, both the engine's own.
        let deadline = Instant::now() + Duration::from_millis(50);
        rt2.set_interrupt_handler(Some(Box::new(move || Instant::now() > deadline)));
        rt2.set_memory_limit(8 * 1024 * 1024);
        let t6 = Instant::now();
        let r: Result<i32, _> = ctx2.with(|ctx| ctx.eval("while(true){}"));
        let t7 = Instant::now();
        println!("quickjs\twhile(true) under 50 ms interrupt: {:?} after {:?}", r.map(|_| "finished").map_err(|e| e.to_string()), t7 - t6);
        rt2.set_interrupt_handler(None);
        let t8 = Instant::now();
        let r: Result<i32, _> = ctx2.with(|ctx| ctx.eval("var a = new Array(5000000).fill(0); a.length"));
        let t9 = Instant::now();
        println!("quickjs\tnew Array(5e6).fill(0) under 8 MiB: {:?} after {:?}", r.map(|_| "finished").map_err(|e| e.to_string()), t9 - t8);
    }

        println!("duktape\tconstruct={:?}\teval(1+1)={:?}\teval(loop1000)={:?}", t1 - t0, t2 - t1, t3 - t2);

## src/bin/v8.rs

    //! Initialise V8, create an isolate and context, evaluate `1+1`, and time each stage.
    use std::time::Instant;
    fn main() {
        let t0 = Instant::now();
        let platform = v8::new_default_platform(0, false).make_shared();
        v8::V8::initialize_platform(platform);
        v8::V8::initialize();
        let t1 = Instant::now();
        let isolate = &mut v8::Isolate::new(v8::CreateParams::default());
        v8::scope!(let handle_scope, isolate);
        let context = v8::Context::new(handle_scope, Default::default());
        let scope = &v8::ContextScope::new(handle_scope, context);
        let t2 = Instant::now();
        let code = v8::String::new(scope, "1+1").unwrap();
        let script = v8::Script::compile(scope, code, None).unwrap();
        let result = script.run(scope).unwrap();
        let t3 = Instant::now();
        assert_eq!(result.int32_value(scope), Some(2));
        println!("v8\tplatform+initialize={:?}\tisolate+context={:?}\teval(1+1)={:?}", t1 - t0, t2 - t1, t3 - t2);
    }
